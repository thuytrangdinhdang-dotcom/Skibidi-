local autoTeleBase = false

local function teleportToBase()
    local char = LocalPlayer.Character
    if not char then return end
    char:PivotTo(CFrame.new(BASE_POSITION_TP))
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if hrp then
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end
end

local function getNearestPrompt()
    local char = LocalPlayer.Character
    if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end

    local nearestPrompt
    local nearestDistance = PROMPT_DISTANCE

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") and obj.Enabled then
            local position
            if obj.Parent:IsA("BasePart") then
                position = obj.Parent.Position
            elseif obj.Parent:IsA("Attachment") then
                position = obj.Parent.WorldPosition
            elseif obj.Parent:IsA("Model") then
                local primary = obj.Parent.PrimaryPart
                if primary then
                    position = primary.Position
                end
            end

            if position then
                local distance = (hrp.Position - position).Magnitude
                if distance <= nearestDistance then
                    nearestDistance = distance
                    nearestPrompt = obj
                end
            end
        end
    end
    return nearestPrompt
end

local function activateNearestPrompt()
    if not autoTeleBase then return end
    local prompt = getNearestPrompt()
    if not prompt then return end
    prompt.HoldDuration = 0

    pcall(function()
        prompt.MaxActivationDistance = math.max(
            prompt.MaxActivationDistance,
            PROMPT_DISTANCE
        )
    end)

    if typeof(fireproximityprompt) == "function" then
        pcall(function() fireproximityprompt(prompt) end)
    else
        pcall(function()
            prompt:InputHoldBegin()
            prompt:InputHoldEnd()
        end)
    end
end

local function handleRagdoll()
    if not autoTeleBase then return end
    if isTeleporting then return end
    isTeleporting = true
    task.spawn(function()
        activateNearestPrompt()
        task.wait(PROMPT_TO_BASE_DELAY)
        if autoTeleBase then teleportToBase() end
        isTeleporting = false
    end)
end

RunService.Heartbeat:Connect(function()
    if not autoTeleBase then
        hasRecovered = true
        return
    end
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local state = hum:GetState()
    local isRagdoll =
        state == Enum.HumanoidStateType.Physics
        or state == Enum.HumanoidStateType.Ragdoll
        or state == Enum.HumanoidStateType.FallingDown

    if not isRagdoll then
        hasRecovered = true
        return
    end

    if isRagdoll and hasRecovered and not isTeleporting then
        hasRecovered = false
        handleRagdoll()
    end
end)

local function makeNormalDrag(gui)
    local dragging = false
    local dragStart
    local startPos

    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            dragStart = input.Position
            startPos = gui.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
end

utility.GetConnections = function(obj, signal)
    local s, r = pcall(function(...) return getconnections(obj[signal]) end)
    if s and r then return r end
    warn("failed to getconnections error: "..tostring(r))
    return nil
end

utility.Disconnect = function(conns)
    local s, r = pcall(function(...)
        local patched = 0
        for _, conn in next, conns do
            conn:Disconnect()
            patched +=1
        end
        return patched
    end)
    if s and r ~= 0 then return warn("patched: "..tostring(r).. " connections") end
    return warn("patched nothing: "..tostring(r))
end

local autoBuyUtility = {
    ReplicatedStorage = ReplicatedStorage,
    IsRunning = false,
    Thread = nil,
}

function autoBuyUtility:GetTreadmils()
    local s, r = pcall(function(...)
        local d = self.Save.Get()
        local o = {}
        local l = d.TreadmillUpgradeLevel
        for id, t in next, self.Treadmills.Directory do
            if self.Treadmills.GetUpgradeLevel(id) > l and self.CanAfford(d.Money, t.Price) then
                table.insert(o, id)
            end
        end
        return o
    end)
    if s and r then return r end
    return nil
end

function autoBuyUtility:GetBaseUpgradeable()
    local s, r = pcall(function(...)
        local d = self.Save.Get()
        local l = d.BaseUpgradeLevel + 1
        local n = self.Bases.BASES[l]
        if self.CanAfford(d.Money, n.Cost) then return true end
        return false
    end)
    if s and r then return r end
    return false
end

function autoBuyUtility:GetTrails()
    local s, r = pcall(function(...)
        local d = self.Save.Get()
        local o = {}
        for _, trial in next, self.Trails.Directory do
            if not d.TrailInventory[trial._id] and self.CanAfford(d.Money, trial.Price) then
                table.insert(o, trial._id)
            end
        end
        return o
    end)
    if s and r then return r end
    return nil
end

autoBuyUtility.CanAfford = function(m, c)
    local s, r = pcall(function(...) return m >= c end)
    if s and r then return r end
    return false
end

function autoBuyUtility:Purchase(remote, id)
    local s, r = pcall(function(...) return remote:InvokeServer(id) end)
    if s and r then return r end
    return false
end

function autoBuyUtility:Init()
    self.Data = self.ReplicatedStorage:FindFirstChild("Data")
    if not self.Data then return warn("Couldnt get data") end
    self.Treadmills = require(self.Data:FindFirstChild("Treadmills"))
    if not self.Treadmills then return warn("failed to get Treadmills") end
    self.Trails = require(self.Data:FindFirstChild("Trails"))
    if not self.Trails then return warn("Couldnt get Trails") end
    self.Bases = require(self.Data:FindFirstChild("Bases"))
    if not self.Bases then return warn("Couldnt get Bases") end
    self.Shared = self.ReplicatedStorage:FindFirstChild("Shared")
    if not self.Shared then return warn("couldnt get Shared") end
    self.Save = require(self.Shared:FindFirstChild("Save"))
    if not self.Save then return warn("couldnt get Shared") end
    self.Packages = self.ReplicatedStorage:FindFirstChild("Packages")
    if not self.Packages then return warn("couldnt get Packages") end
    self.Networking = self.Packages:FindFirstChild("Networking")
    if not self.Networking then return warn("couldnt get Networking") end
    self["RF/Trailwear/AskPurchase"] = self.Networking:FindFirstChild("RF/Trailwear/AskPurchase")
    if not self["RF/Trailwear/AskPurchase"] then return warn("couldnt get RF/Trailwear/AskPurchase") end
    self["RF/Treadmill/AskTierRaise"] = self.Networking:FindFirstChild("RF/Treadmill/AskTierRaise")
    if not self["RF/Treadmill/AskTierRaise"] then return warn("couldnt get RF/Treadmill/AskTierRaise") end
    self["RE/Homestead/AskBaseTierRaise"] = self.Networking:FindFirstChild("RE/Homestead/AskBaseTierRaise")
    if not self["RE/Homestead/AskBaseTierRaise"] then return warn("couldnt get RE/Homestead/AskBaseTierRaise") end
    return true
end

function autoBuyUtility:Start()
    if self.IsRunning then return end
    if not self:Init() then return warn("AutoBuy init failed") end
    self.IsRunning = true
    self.Thread = task.spawn(function()
        while self.IsRunning do
            task.wait(0.1)
            local trials = self:GetTrails()
            if typeof(trials) == "table" then
                for _, trail in next, trials do
                    self:Purchase(self["RF/Trailwear/AskPurchase"], trail)
                    task.wait(0.1)
                end
            end
            local treadmils = self:GetTreadmils()
            if typeof(treadmils) == "table" then
                for _, treadmil in next, treadmils do
                    self:Purchase(self["RF/Treadmill/AskTierRaise"], treadmil)
                    task.wait(0.1)
                end
            end
            local upgradeable = self:GetBaseUpgradeable()
            if upgradeable then
                self["RE/Homestead/AskBaseTierRaise"]:FireServer()
            end
        end
    end)
end

function autoBuyUtility:Stop()
    self.IsRunning = false
    if self.Thread then
        task.cancel(self.Thread)
        self.Thread = nil
    end
end

local function FormatSpeedNumber(val)
    local num = tonumber(val)
    if not num then return tostring(val or "N/A") end
    if num >= 1e12 then return string.format("%.2fT", num / 1e12)
    elseif num >= 1e9 then return string.format("%.2fB", num / 1e9)
    elseif num >= 1e6 then return string.format("%.2fM", num / 1e6)
    elseif num >= 1e3 then return string.format("%.2fK", num / 1e3)
    else return string.format("%d", math.floor(num)) end
end

local function NormalizeName(value) return string.lower(tostring(value or "")) end

local function ContainsKeyword(text, keywordsList)
    text = NormalizeName(text)
    for _, keyword in ipairs(keywordsList) do
        if string.find(text, keyword, 1, true) then return true end
    end
    return false
end

local function IsTrapObject(object)
    if not object or object == Workspace then return false end
    if ContainsKeyword(object.Name, TRAP_WORDS) then return true end
    for _, attributeName in ipairs({ "Trap", "Type", "ObjectType", "ItemType", "Name", "DisplayName" }) do
        local success, value = pcall(function() return object:GetAttribute(attributeName) end)
        if success and value ~= nil and ContainsKeyword(value, TRAP_WORDS) then return true end
    end
    return false
end

    local guardName = object.Name
    local zoneName = "Unknown Zone"
    local color = Color3.fromRGB(255, 165, 0)

    local current = object.Parent  
    while current and current ~= Workspace do  
        local lowerCName = current.Name:lower()  
        if lowerCName:find("zone") or lowerCName:find("area") or lowerCName:find("map") or lowerCName:find("spawn") or lowerCName:find("base") then  
            zoneName = current.Name  
            break  
        end  
        current = current.Parent  
    end  

    local combined = (guardName .. " " .. zoneName):lower()  
    if combined:find("zone1") or combined:find("khu 1") or combined:find("easy") then color = Color3.fromRGB(0, 255, 255)  
    elseif combined:find("zone2") or combined:find("khu 2") or combined:find("medium") then color = Color3.fromRGB(0, 255, 0)  
    elseif combined:find("zone3") or combined:find("khu 3") or combined:find("hard") then color = Color3.fromRGB(255, 255, 0)  
    elseif combined:find("zone4") or combined:find("khu 4") or combined:find("extreme") then color = Color3.fromRGB(255, 128, 0)  
    elseif combined:find("boss") or combined:find("vip") then color = Color3.fromRGB(255, 0, 255)  
    else  
        local hash = 0  
        for i = 1, #guardName do hash = hash + string.byte(guardName, i) end  
        color = Color3.fromRGB((hash * 37) % 200 + 55, (hash * 47) % 200 + 55, (hash * 57) % 200 + 55)  
    end  
    return guardName, zoneName, color
end

local function ScanAllObjects()
    for _, object in ipairs(Workspace:GetDescendants()) do
        if IsTrapObject(object) then
            TrapCache[object] = true
            if Config.EspTrap then CreateTrapESP(object) end
            if Config.DeleteTrap then pcall(function() object:Destroy() end) end
        end
        if IsParasiteObject(object) then
            ParasiteCache[object] = true
            if Config.EspParasite then CreateParasiteESP(object) end
        end
        if IsGuardObject(object) then
            GuardCache[object] = true
            if Config.EspGuard then CreateGuardESP(object) end
        end
        if IsTreadmillObject(object) then
            TreadmillCache[object] = true
            if Config.DeleteTreadmill then pcall(function() object:Destroy() end) end
        end
    end
end

local function IsEggPrompt(prompt)
    if not prompt or not prompt:IsA("ProximityPrompt") then return false end
    local nameText = string.lower(tostring(prompt.Name) .. " " .. tostring(prompt.ObjectText))
    local actionText = string.lower(tostring(prompt.ActionText))
    local nameMatch = string.find(nameText, "trứng", 1, true) or string.find(nameText, "trung", 1, true) or string.find(nameText, "egg", 1, true)
    local actionMatch = string.find(actionText, "trộm", 1, true) or string.find(actionText, "trom", 1, true) or string.find(actionText, "cắp", 1, true) or string.find(actionText, "steal", 1, true)
    return nameMatch ~= nil or actionMatch ~= nil
end

local function GetPlayerSpeedFromLeaderboard(player)

local function UpdatePlayerESP()
    local myChar = GetCharacter()
    local myRoot = GetRoot(myChar)
    for player, data in pairs(PlayerESPObjects) do
        if player and player.Parent and player.Character and player.Character:FindFirstChild("Head") then
            local targetChar = player.Character
            local targetRoot = GetRoot(targetChar)
            local targetHum = GetHumanoid(targetChar)
            local distStr = "N/A"
            if myRoot and targetRoot then
                local dist = math.floor((myRoot.Position - targetRoot.Position).Magnitude)
                distStr = tostring(dist) .. "m"
            end
            local speedStr = GetPlayerSpeedFromLeaderboard(player)

            if data.Label then  
                data.Label.Text = string.format("%s | [%s]\nSpeed: %s", player.DisplayName or player.Name, distStr, speedStr)  
            end  
            if data.HealthBar and targetHum then  
                local hpPercent = math.clamp(targetHum.Health / math.max(targetHum.MaxHealth, 1), 0, 1)  
                data.HealthBar.Size = UDim2.fromScale(hpPercent, 1)  
                data.HealthBar.BackgroundColor3 = Color3.fromRGB(255 * (1 - hpPercent), 255 * hpPercent, 50)  
            end  
        else  
            RemovePlayerESP(player)  
        end  
    end
end

local function ReplaceHumanoid(character)
    if not Config.Godmode then return end
    local oldHumanoid = character:WaitForChild("Humanoid", 5)
    if not oldHumanoid then return end
    local data = {}
    for _, prop in ipairs(HUM_PROPERTIES) do
        local ok, val = pcall(function() return oldHumanoid[prop] end)
        if ok then data[prop] = val end
    end
    local walkSpeed = GetAdaptiveSpeed()
    pcall(function() oldHumanoid:Destroy() end)
    RunService.Heartbeat:Wait()
    local newHumanoid = Instance.new("Humanoid")
    for prop, val in pairs(data) do
        pcall(function() newHumanoid[prop] = val end)
    end
    newHumanoid.Parent = character
    RunService.Heartbeat:Wait()
    newHumanoid.WalkSpeed = walkSpeed
    pcall(function() newHumanoid:ChangeState(Enum.HumanoidStateType.Landed) end)
    local animateScript = character:FindFirstChild("Animate")
    if animateScript then
        animateScript.Enabled = false
        task.spawn(function()
            task.wait(0.05)
            if animateScript and animateScript.Parent then animateScript.Enabled = true end
        end)
    end
    local camera = Workspace.CurrentCamera
    if camera then pcall(function() camera.CameraSubject = newHumanoid end) end
end

local function ForceAntiRagdollReset()
    local char = GetCharacter()
    if not char then return end
    local hum = GetHumanoid(char)
    local root = GetRoot(char)
    if not hum or not root or hum.Health <= 0 then return end
    pcall(function()
        hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        root.Velocity = Vector3.zero
        root.RotVelocity = Vector3.zero
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        for _, obj in ipairs(char:GetDescendants()) do
            if obj:IsA("Motor6D") or obj:IsA("Constraint") then obj.Enabled = true end
        end
        if Workspace.CurrentCamera then Workspace.CurrentCamera.CameraSubject = hum end
        hum.AutoRotate = true; hum.PlatformStand = false; hum.Sit = false
    end)
end

local function isBat(tool)
    if not tool or not tool:IsA("Tool") then return false end
    local name = tool.Name:lower()
    for _, keyword in ipairs(BAT_NAMES) do
        if name:find(keyword, 1, true) then return true end
    end
    return false
end

local function getBat()
    local character = GetCharacter()
    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    if character then
        for _, object in ipairs(character:GetChildren()) do if isBat(object) then return object end end
    end
    if backpack then
        for _, object in ipairs(backpack:GetChildren()) do if isBat(object) then return object end end
    end
    return nil
end

local function equipBat()
    local character = GetCharacter()
    if not character then return nil end
    local humanoid = GetHumanoid(character)
    if not humanoid then return nil end
    local bat = getBat()
    if not bat then return nil end
    if bat.Parent ~= character then
        pcall(function() humanoid:EquipTool(bat) end)
        task.wait()
    end
    return bat
end

local function getNearestEnemy(maxRange)
    local character = GetCharacter()
    if not character then return nil end
    local root = GetRoot(character)
    if not root then return nil end
    local nearestCharacter = nil
    local nearestDistance = maxRange or ATTACK_RANGE
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local targetCharacter = player.Character
            if targetCharacter then
                local targetRoot = GetRoot(targetCharacter)
                local targetHumanoid = GetHumanoid(targetCharacter)
                if targetRoot and targetHumanoid and targetHumanoid.Health > 0 then
                    local distance = (targetRoot.Position - root.Position).Magnitude
                    if distance <= nearestDistance then
                        nearestDistance = distance
                        nearestCharacter = targetCharacter
                    end
                end
            end
        end
    end
    return nearestCharacter
end

local function attackBat(maxRange)
    local target = getNearestEnemy(maxRange)
    if not target then return end
    local bat = equipBat()
    if not bat then return end
    pcall(function() bat:Activate() end)
end

local function GetEggValueScore(prompt)
    local text = string.lower(tostring(prompt.Name) .. " " .. tostring(prompt.ObjectText) .. " " .. tostring(prompt.ActionText))
    local parentObj = prompt.Parent
    while parentObj and parentObj ~= Workspace do
        text = text .. " " .. string.lower(parentObj.Name)
        for _, attr in ipairs({"Name", "DisplayName", "Type", "Rarity", "Value"}) do
            local success, val = pcall(function() return parentObj:GetAttribute(attr) end)
            if success and val then text = text .. " " .. string.lower(tostring(val)) end
        end
        parentObj = parentObj.Parent
    end

    local priorityKeywords = { {"eternal", 999999999999}, {"secret", 99999999999}, {"cosmic", 9999999999}, {"mythic", 999999999} }  
    local score = 0  
    for _, pair in ipairs(priorityKeywords) do  
        if string.find(text, pair[1], 1, true) then score = pair[2]; break end  
    end  
    for word in string.gmatch(text, "%$(%d+)") do  
        local num = tonumber(word)  
        if num then score = score + num * 10 end  
    end  
    return score
end

local function FindBestEggPrompt()
    local bestPrompt = nil
    local maxScore = -1
    for prompt in pairs(EggPromptCache) do
        if prompt and prompt.Parent then
            local score = GetEggValueScore(prompt)
            if score > maxScore then
                maxScore = score
                bestPrompt = prompt
            end
        else
            EggPromptCache[prompt] = nil
        end
    end
    return bestPrompt
end

local function FirePrompt(prompt)
    if not prompt or not prompt:IsA("ProximityPrompt") or not prompt.Parent then return end
    TargetPrompt = prompt
    pcall(function()
        prompt.MaxActivationDistance = 999999
        prompt.RequiresLineOfSight = false
        prompt.HoldDuration = 0
        prompt.Enabled = true
        if fireproximityprompt then fireproximityprompt(prompt) end
    end)
end

local function FlyToPosition(targetPosition)
    local char = GetCharacter()
    if not char then return end
    local root = GetRoot(char)
    if not root then return end
    EnableNoclip()  

    local flyHeight = Config.FlightHeight  
    local targetPosWithHeight = Vector3.new(targetPosition.X, flyHeight, targetPosition.Z)  
    local connection  
    connection = RunService.Heartbeat:Connect(function()  
        if not Config.AutoStealBest or not char.Parent or not root.Parent then  
            if connection then connection:Disconnect() end  
            return  
        end  
        local currentPos = root.Position  
        local direction = (targetPosWithHeight - currentPos)  
        local distance = direction.Magnitude  
        if distance < 4 then  
            root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)  
            if connection then connection:Disconnect() end  
            return  
        end  
        local moveSpeed = GetAdaptiveSpeed()  
        root.AssemblyLinearVelocity = direction.Unit * moveSpeed  
        root.CFrame = CFrame.new(currentPos, Vector3.new(targetPosWithHeight.X, currentPos.Y, targetPosWithHeight.Z))  
    end)  

    while Config.AutoStealBest and char.Parent and root.Parent do  
        if (root.Position - targetPosWithHeight).Magnitude < 4 then break end  
        task.wait(0.05)  
    end  
    if connection then connection:Disconnect() end  
    if root and root.Parent then root.AssemblyLinearVelocity = Vector3.new(0, 0, 0) end
end

local function StartAutoStealLoop()
    DisconnectConnection("AutoStealLoop")
    table.clear(EggPromptCache)
    for _, object in ipairs(Workspace:GetDescendants()) do
        if object:IsA("ProximityPrompt") and IsEggPrompt(object) then
            EggPromptCache[object] = true
            ApplyInstantEggPrompt(object)
        end
    end

    Connections.AutoStealLoop = task.spawn(function()  
        while Config.AutoStealBest do  
            local bestEgg = FindBestEggPrompt()  
            if bestEgg and bestEgg.Parent then  
                local eggPos = nil  
                if bestEgg.Parent:IsA("BasePart") then eggPos = bestEgg.Parent.Position  
                elseif bestEgg.Parent:IsA("Model") then pcall(function() eggPos = bestEgg.Parent:GetPivot().Position end) end  

                if eggPos then  
                    FlyToPosition(eggPos)  
                    local char = GetCharacter()  
                    local root = GetRoot(char)  
                    local startTime = tick()  
                    while root and (root.Position - Vector3.new(eggPos.X, Config.FlightHeight, eggPos.Z)).Magnitude > 5 and (tick() - startTime) < 3 and Config.AutoStealBest do  
                        task.wait(0.05)  
                        root = GetRoot(GetCharacter())  
                    end  
                    local attempt = 0  
                    while bestEgg and bestEgg.Parent and attempt < 5 and Config.AutoStealBest do  
                        FirePrompt(bestEgg)  
                        task.wait(0.15)  
                        attempt = attempt + 1  
                    end  
                    if bestEgg and not bestEgg.Parent then EggPromptCache[bestEgg] = nil end  
                    task.wait(0.1)  
                    FlyToPosition(Config.BASE_POS)  
                end  
            end  
            task.wait(0.8)  
        end  
        DisableNoclip()  
    end)
end

    DisconnectConnection("AutoStealLoop")
    DisableNoclip()
end

local VisualPet = { Container = nil, Spawned = {}, OrbitRadius = 4, OrbitSpeed = 0.8, OrbitConnection = nil, PetNames = {}, SelectedName = "" }
local function CreateContainer()
    local folder = Instance.new("Folder")
    folder.Name = "VxzVisualPets"; folder.Parent = workspace
    return folder
end
VisualPet.Container = CreateContainer()

function VisualPet:Scan()
    local nameMap = {}
    local results = {}
    local function checkModel(model)  
        if not model:IsA("Model") then return end  
        local lower = model.Name:lower()  
        if lower:find("humanoid") or lower:find("egg") then return end  
        if not model:FindFirstChildWhichIsA("BasePart") then return end  
        if model:IsDescendantOf(self.Container) then return end  
        local parentName = model.Parent and model.Parent.Name:lower() or ""  
        if lower:find("pet") or parentName:find("pet") or parentName:find("pen") or parentName:find("render") or parentName:find("slot") then  
            if not nameMap[model.Name] then  
                nameMap[model.Name] = true  
                table.insert(results, model.Name)  
            end  
        end  
    end  
    for _, desc in ipairs(workspace:GetDescendants()) do checkModel(desc) end  
    table.sort(results)  
    self.PetNames = results  
    return results
end

function VisualPet:Find(name)
    local search = name:lower()
    for _, desc in ipairs(workspace:GetDescendants()) do
        if desc:IsA("Model") and desc.Name:lower():find(search, 1, true) and desc:FindFirstChildWhichIsA("BasePart") and not desc:IsDescendantOf(self.Container) then
            return desc
        end
    end
    return nil
end

function VisualPet:Spawn(name)
    local original = self:Find(name)
    if not original then return end
    local clone = original:Clone()  
    for _, child in ipairs(clone:GetDescendants()) do  
        if child:IsA("Script") or child:IsA("LocalScript") or child:IsA("ModuleScript") then child:Destroy() end  
        if child:IsA("BasePart") then  
            child.Anchored = true; child.CanCollide = false; child.CanTouch = false; child.CastShadow = false  
        end  
    end  
    clone.Name = "VP_" .. name  
    clone.Parent = self.Container  
    local angle = #self.Spawned * (2 * math.pi / math.max(1, #self.Spawned + 1))  
    table.insert(self.Spawned, { model = clone, angle = angle })
end

function VisualPet:RemoveLast()
    if #self.Spawned == 0 then return end
    local data = table.remove(self.Spawned)
    pcall(function() data.model:Destroy() end)
end

function VisualPet:RemoveAll()
    for _, data in ipairs(self.Spawned) do pcall(function() data.model:Destroy() end) end
    self.Spawned = {}
end

function VisualPet:StartOrbit()
    if self.OrbitConnection then return end
    self.OrbitConnection = RunService.Heartbeat:Connect(function()
        local rootPart = GetRoot(GetCharacter())
        if not rootPart or #self.Spawned == 0 then return end
        local center = rootPart.Position + Vector3.new(0, 1.5, 0)  
        local count = #self.Spawned  
        local time = tick()  
        for i, data in ipairs(self.Spawned) do  
            local angle = (i - 1) * (2 * math.pi / count) + time * self.OrbitSpeed  
            local pos = center + Vector3.new(math.cos(angle) * self.OrbitRadius, 0, math.sin(angle) * self.OrbitRadius)  
            local model = data.model  
            if model and model.Parent then  
                pcall(function() model:PivotTo(CFrame.new(pos) * CFrame.Angles(0, angle + math.pi, 0)) end)  
            end  
        end  
    end)
end

function VisualPet:StopOrbit()
    if self.OrbitConnection then
        self.OrbitConnection:Disconnect()
        self.OrbitConnection = nil
    end
end

    Callback = function(text) if text and text ~= "" then VisualPet.SelectedName = text end end
})
VisualPetTab:Slider({
    Title = "Bán kính orbit", Desc = "Khoảng cách pet so với bạn", Step = 1, Value = { Min = 2, Max = 20, Default = 4 },
    Callback = function(val) VisualPet.OrbitRadius = val end
})
VisualPetTab:Slider({
    Title = "Tốc độ quay", Desc = "0 = đứng yên, 10 = nhanh nhất", Step = 1, Value = { Min = 0, Max = 10, Default = 1 },
    Callback = function(val) VisualPet.OrbitSpeed = val * 0.2 end
})
VisualPetTab:Button({
    Title = " Spawn Pet", Desc = "spawn model", Color = Color3.fromRGB(48, 255, 106),
    Callback = function()
        if VisualPet.SelectedName == "" then return end
        VisualPet:Spawn(VisualPet.SelectedName)
        if not VisualPet.OrbitConnection then VisualPet:StartOrbit() end
    end
})
VisualPetTab:Button({
    Title = " Remove Last Pet",
    Callback = function()
        VisualPet:RemoveLast()
        if #VisualPet.Spawned == 0 then VisualPet:StopOrbit() end
    end
})
VisualPetTab:Button({
    Title = " Remove All Pets", Color = Color3.fromRGB(255, 72, 48),
    Callback = function()
        VisualPet:RemoveAll()
        VisualPet:StopOrbit()
    end
})

task.delay(3, function()
    local pets = VisualPet:Scan()
    if #pets > 0 then
        pcall(function()
            petDropdown:Refresh(pets)
            petDropdown:SetValues(pets)
            VisualPet.SelectedName = pets[1] or ""
        end)
    end
end)

local function CleanupVisualPet()
    VisualPet:RemoveAll()
    VisualPet:StopOrbit()
    pcall(function() VisualPet.Container:Destroy() end)
end
pcall(function()
    game:GetService("RunService").Stepped:Connect(function()
        if not getgenv or not getgenv().Loaded then CleanupVisualPet() end
    end)
end)

ScanAllObjects()
ScanAllEggPrompts()

Workspace.DescendantAdded:Connect(function(object)
    if IsTrapObject(object) then
        if Config.DeleteTrap then DeleteTrapObject(object) end
        if Config.EspTrap then CreateTrapESP(object) end
    elseif IsTreadmillObject(object) then
        if Config.DeleteTreadmill then DeleteTreadmillObject(object) end
    elseif IsParasiteObject(object) then
        if Config.EspParasite then CreateParasiteESP(object) end
    elseif IsGuardObject(object) then
        if Config.EspGuard then CreateGuardESP(object) end
    end
    if Config.AutoStealBest and object:IsA("ProximityPrompt") and IsEggPrompt(object) then
        EggPromptCache[object] = true
    end
end)
